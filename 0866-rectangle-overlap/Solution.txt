class Solution {
public:
    bool isRectangleOverlap(vector<int>& rec1, vector<int>& rec2) {
        int x0 = rec1[0], x1 = rec1[2];
        int y0 = rec1[1], y1 = rec1[3];

        int x2 = rec2[0], x3 = rec2[2];
        int y2 = rec2[1], y3 = rec2[3];

        if (x0 == x1 || y0 == y1 || x2 == x3 || y2 == y3)
            return false;

        return !(x1 <= x2 || y1 <= y2 || x0 >= x3 || y0 >= y3);
    }
};
/*
   L - x1 , y1
   U - x2, y2
