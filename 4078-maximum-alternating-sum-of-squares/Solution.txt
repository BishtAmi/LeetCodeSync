class Solution {
public:
    long long maxAlternatingSum(vector<int>& nums) {
        int n = nums.size();
        for (int i = 0; i < n; i++) {
            nums[i] = (nums[i] < 0) ? -nums[i] : nums[i];
        }
        sort(nums.rbegin(), nums.rend());
        long long sum = 0;
        for (int i = 0; i < n; i++) {
            if (i < (n + 1) / 2) {
                sum += 1LL * nums[i] * nums[i];
            } else
                sum -= 1LL * nums[i] * nums[i];
        }
        return sum;
    }
};
