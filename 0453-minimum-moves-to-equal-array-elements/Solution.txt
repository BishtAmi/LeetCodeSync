class Solution {
public:
    int minMoves(vector<int>& nums) {
        int n = nums.size();
        int mini = *min_element(nums.begin(),nums.end());
        int sum = 0;
        for(auto x:nums)
        {
            sum += x;
        }
        return sum - mini * n;
    }
};
/*

 S + k(n-1) = ( mini + k ) * n
 kn - k = mini * n + kn - S

 k = S - mini * n 
*/
