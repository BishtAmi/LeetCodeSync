class Solution {
public:
    int candy(vector<int>& ratings) {
        int n = ratings.size();
        vector<int> candi(n, 1);
        for (int i = 0; i < n; i++) {
            int cur = candi[i];
            if (i + 1 < n && ratings[i] > ratings[i + 1]) {
                if (cur <= candi[i])
                    cur = candi[i + 1] + 1;
            }
            if (i - 1 >= 0 && ratings[i] > ratings[i - 1]) {
                if (cur <= candi[i - 1])
                    cur = max(cur, candi[i - 1] + 1);
            }
            candi[i] = cur;
        }

        for (int i = n - 1; i >= 0; i--) {
            int cur = candi[i];
            if (i + 1 < n && ratings[i] > ratings[i + 1]) {
                if (cur <= candi[i])
                    cur = candi[i + 1] + 1;
            }
            if (i - 1 >= 0 && ratings[i] > ratings[i - 1]) {
                if (cur <= candi[i - 1])
                    cur = max(cur, candi[i - 1] + 1);
            }
            candi[i] = cur;
        }

        return accumulate(candi.begin(), candi.end(), 0);
    }
};
