class Solution {
public:
    string evaluate(string s, vector<vector<string>>& knowledge) {

        map<string, string> keyToVal;
        for (auto str : knowledge) {
            keyToVal[str[0]] = str[1];
        }
        string cur, ans;
        bool isFound = false;
        for (auto ch : s) {
            if (ch == '(') {
                ans += cur;
                cur = "";
                isFound = true;
                continue;
            }
            if (isFound && ch == ')') {
                isFound = false;
                ans += keyToVal.count(cur) ? keyToVal[cur] : "?";
                cur = "";
                continue;
            }
            cur += ch;
        }
        ans += cur;
        return ans;
    }
};
