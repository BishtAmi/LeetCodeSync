class BusBooking {
public:
    BusBooking(int n) {}

    unordered_map<string, int> freq;
    int total = 0;

    void toggle(string seat) {
        if (freq.find(seat) != freq.end()) {
            total -= freq[seat];
            freq.erase(seat);
        } else {
            if (seat[0] == 'B' || seat[0] == 'C') {
                freq[seat] = 1;
            } else {
                int n = seat.size();
                string b = "B" + seat.substr(1, n - 1);
                string c = "C" + seat.substr(1, n - 1);

                if (seat[0] == 'A') {
                    freq[seat] = (freq.find(b) != freq.end()) ? 3 : 1;
                } else if (seat[0] == 'D') {
                    freq[seat] = (freq.find(c) != freq.end()) ? 3 : 1;
                } else {
                    freq[seat] = 1;
                }
            }
            total += freq[seat];
        }
    }

    int getTotalTime() { return total; }

    int getMinTime() { return freq.size(); }
};
