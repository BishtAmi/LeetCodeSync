class BusBooking {
public:
    BusBooking(int n) {}

    unordered_map<string, int> freq, costToadd;
    set<string> st;
    int total = 0;
    void toggle(string seat) {
        if (freq[seat]) {
            freq[seat] = 0;
            total -= costToadd[seat];
            st.erase(seat);
        } else {
            st.insert(seat);
            if (seat[0] == 'B' || seat[0] == 'C') {
                costToadd[seat] = 1;
            } else {
                int n = seat.size();
                string b = "B";
                b += seat.substr(1, n - 1);
                string c = "C";
                c += seat.substr(1, n - 1);
                if (seat[0] == 'A') {
                    costToadd[seat] = freq[b] ? 3 : 1;
                } else if (seat[0] == 'D') {
                    costToadd[seat] = freq[c] ? 3 : 1;
                } else
                    costToadd[seat] = 1;
            }
            total += costToadd[seat];
            freq[seat] = 1;
        }
    }

    int getTotalTime() { return total; }

    int getMinTime() { return st.size(); }
};

 * Your BusBooking object will be instantiated and called as such:
 * BusBooking* obj = new BusBooking(n);
 * obj->toggle(seat);
 * int param_2 = obj->getTotalTime();
 * int param_3 = obj->getMinTime();
 */
