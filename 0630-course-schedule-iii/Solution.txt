class Solution {
public:
    int scheduleCourse(vector<vector<int>>& courses) {
        sort(courses.begin(), courses.end(), [](auto& a, auto& b) {
            if (a[1] == b[1])
                return a[0] < b[0];
            return a[1] < b[1];
        });
        int limit = 0, cnt = 0;
        priority_queue<int> pq;
        for (auto course : courses) {
            limit += course[0];
            pq.push(course[0]);
            if (limit > course[1]) {
                limit -= pq.top();
                pq.pop();
            }
        }
        return pq.size();
    }
};
