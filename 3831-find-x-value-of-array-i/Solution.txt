class Solution {
public:
    

    vector<long long> resultArray(vector<int>& nums, int k) {
        int n = nums.size();
        vector<long long> res(k,0);
        vector<int> cnt(k, 0);
        for (auto num : nums) {
            vector<int> temp(k, 0);
            for (int i = 0; i < k; i++) {
                temp[1LL * i * num % k] += cnt[i];
                res[1LL * i * num % k] += cnt[i];
            }
            cnt = temp;
            cnt[num%k]++;
            res[num%k]++;
        }
        return res;
    }
};
