class Solution {
public:
    int minOperations(vector<int>& nums, vector<int>& locked, int k) {
        int ans = 0;
        int n = nums.size();
        for (int c = 1; c <= n; c++) {
            int cnt = 0, indx = 0;
            for (int i = 0; i < n; i++) {
                cnt += locked[i] == 1;
                locked[i] = 0;
                if (nums[i] == c) {
                    indx = i;
                    if (indx + 1 == c)
                        cnt = 0;
                    break;
                }
            }
            // cout << c << " " << cnt << " " << indx - c + 1 << "\n";
            ans += cnt;
            ans += indx - c + 1;
            if (cnt > k) {
                return -1;
            }
            k -= cnt;
            for (int i = indx; i > 0; i--) {
                swap(nums[i], nums[i - 1]);
            }
        }
        return ans;
    }
};
