class Solution {
public:
    int findTheDistanceValue(vector<int>& arr1, vector<int>& arr2, int d) {
        sort(arr2.begin(), arr2.end());
        int cnt = 0;
        for (auto x : arr1) {
            int end =
                upper_bound(arr2.begin(), arr2.end(), x + d) - arr2.begin();
            int start =
                upper_bound(arr2.begin(), arr2.end(), x - d - 1) - arr2.begin();
            end--;
            cnt += end < start;
        }
        return cnt;
    }
};
