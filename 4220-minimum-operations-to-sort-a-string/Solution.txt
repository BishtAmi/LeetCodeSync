class Solution {
public:
    int minOperations(string s) {
        if (is_sorted(s.begin(), s.end()))
            return 0;
        int n = s.length();
        if (n == 2)
            return -1;
        char mini = s[0], maxi = s[0];
        for (auto ch : s) {
            mini = min(mini, ch);
            maxi = max(maxi, ch);
        }
        if (maxi == s[n - 1] || mini == s[0])
            return 1;
        for (int i = 1; i < n - 1; i++) {
            if (maxi == s[i] || mini == s[i]) {
                return 2;
            }
        }
        return 3;
    }
};
